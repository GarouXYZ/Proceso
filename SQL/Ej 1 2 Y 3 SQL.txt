#Crear base de datos
CREATE DATABASE Biblioteca;
#Usar la base de datos
USE Biblioteca;

#Crear las tablas
CREATE TABLE Libro (
	id_Libro INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    Titulo VARCHAR (40) NOT NULL,
    Editorial VARCHAR (40) NOT NULL,
    Area enum ('Ciencia', 'Literatura', 'Tecnologia', 'Historia') NOT NULL
);

CREATE TABLE Estudiante (
	CI INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    Nombre VARCHAR(40) NOT NULL,
    Direccion VARCHAR(40) NOT NULL,
    Curso enum ('BP', 'BTP', 'Deporte', 'Mecanica') NOT NULL, 
    Edad INT CHECK (Edad >= 18 AND Edad <= 80) NOT NULL
);

CREATE TABLE Autor (
	id_Autor INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    Nombre VARCHAR(40) NOT NULL,
    Nacionalidad VARCHAR(40) NOT NULL
);

CREATE TABLE PRESTAMO (
    id_Prestamo INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    FechaP DATE DEFAULT(CURDATE()),
    FechaD DATE,
    Devuelto ENUM('S', 'N') DEFAULT 'N'
);

#Agregas las claves foraneas
ALTER TABLE Libro ADD COLUMN id_Autor INT;
ALTER TABLE Libro ADD CONSTRAINT fk_id_Autor
FOREIGN KEY (id_Autor)references Autor (id_Autor);

ALTER TABLE Prestamo ADD COLUMN id_Libro INT;
ALTER TABLE Prestamo ADD CONSTRAINT fk_id_Libro
FOREIGN KEY (id_Libro)references Libro (id_Libro);

ALTER TABLE Prestamo ADD COLUMN CI INT;
ALTER TABLE Prestamo ADD CONSTRAINT fk_CI
FOREIGN KEY (CI)references Estudiante (CI);

#Crear tabla con 1 clave primaria compuesta de 2 foraneas
CREATE TABLE Libros_Autores (
	id_Libro INT,
    id_Autor INT,

    PRIMARY KEY (id_Libro, id_Autor),
    
	FOREIGN KEY (id_Libro) REFERENCES Libro(id_Libro),
    FOREIGN KEY (id_Autor) REFERENCES Autor(id_Autor)
);
